class TestDirCandidatesPreferredEqualsLegacy:
    """Cover L128: when preferred slug == legacy slug, return single-element tuple."""

    def test_simple_alphanumeric_org_id_returns_single_candidate(self):
        # A slug with only alphanumeric/underscore/hyphen chars where
        # snapshot_slug(org_id) == org_report_snapshot_dir_key(org_id) is impossible
        # because dir_key always appends __<hash>.  But if the legacy slug
        # equals the preferred key (they never will with the hash suffix),
        # the branch returns (preferred,).
        # Instead, test that the tuple length is correct for both cases.
        from cja_auto_sdr.org.snapshot_utils import org_report_snapshot_dir_candidates, snapshot_slug, org_report_snapshot_dir_key

        # The legacy slug can never equal the preferred key (which has __<hash>)
        # so we verify the 2-tuple path is always taken for normal org IDs.
        candidates = org_report_snapshot_dir_candidates("simple_org")
        assert len(candidates) == 2
        assert candidates[0] == org_report_snapshot_dir_key("simple_org")
        assert candidates[1] == snapshot_slug("simple_org")

    def test_legacy_slug_matching_preferred_returns_single(self):
        # Force the condition by monkeypatching
        from unittest.mock import patch
        with patch("cja_auto_sdr.org.snapshot_utils.org_report_snapshot_dir_key", return_value="same"):
            with patch("cja_auto_sdr.org.snapshot_utils.snapshot_slug", return_value="same"):
                from cja_auto_sdr.org.snapshot_utils import org_report_snapshot_dir_candidates
                result = org_report_snapshot_dir_candidates("anything")
                assert result == ("same",)
